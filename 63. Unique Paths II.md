```python
class Solution:
    def uniquePathsWithObstacles(self, obstacleGrid: List[List[int]]) -> int:
        num_rows = len(obstacleGrid)
        num_columns = len(obstacleGrid[0])
        # 間違い unique_paths = [[0] * (num_columns)] * (num_rows)
        unique_paths = [[0 for _ in range(num_columns)] for _ in range(num_rows)]
        if obstacleGrid[0][0] == 0:
            unique_paths[0][0] = 1 # start地点にもある障害物あるのか
        for row in range(1, num_rows):
            if obstacleGrid[row][0] == 1:
                unique_paths[row][0] = 0
                continue
            unique_paths[row][0] = unique_paths[row-1][0]
        for column in range(1, num_columns):
            if obstacleGrid[0][column] == 1:
                unique_paths[0][column] = 0
                continue
            unique_paths[0][column] = unique_paths[0][column-1] 

        for row in range(1, num_rows):
            for column in range(1, num_columns):
                if obstacleGrid[row][column] == 1:
                    unique_paths[row][column] = 0
                    continue
                unique_paths[row][column] = unique_paths[row-1][column] + unique_paths[row][column-1]
        return unique_paths[-1][-1]
```

```python
OBSTACLE = 1

class Solution:
    def uniquePathsWithObstacles(self, obstacleGrid: List[List[int]]) -> int:
        assert obstacleGrid
        assert obstacleGrid[0]
        if obstacleGrid[0][0] == OBSTACLE:
            return 0
        num_rows = len(obstacleGrid)
        num_columns = len(obstacleGrid[0])
        unique_paths = [0] * num_columns
        unique_paths[0] = 1
        for y in range(num_rows):
            dead_end = True
            for x in range(num_columns):
                if obstacleGrid[y][x] == OBSTACLE:
                    unique_paths[x] = 0
                    continue
                dead_end = False
                if x == 0:
                    continue
                unique_paths[x] += unique_paths[x-1]
            if dead_end:
                return 0
        return unique_paths[-1]
```
