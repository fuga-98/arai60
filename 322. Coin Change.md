```python
class Solution:
    def coinChange(self, coins: List[int], amount: int) -> int:
        min_change = [math.inf] * (amount + 1)
        min_change[0] = 0
        for coin in coins:
            for i in range(1, len(min_change)):
                if i < coin:
                    continue
                min_change[i] = min(min_change[i], min_change[i-coin] + 1)
        if min_change[-1] == math.inf:
            return -1
        return min_change[-1]
```

```python
class Solution:
    def coinChange(self, coins: List[int], amount: int) -> int:
        if amount == 0:
            return 0
        possible_amounts = [0]
        num_coin = 0
        checked = set()
        while possible_amounts:
            next_possible_amounts = []
            num_coin += 1
            for coin in coins:
                for value in possible_amounts:
                    next_amount = coin + value
                    if next_amount == amount:
                        return num_coin
                    if next_amount > amount:
                        continue
                    if next_amount in checked:
                        continue
                    checked.add(next_amount)
                    next_possible_amounts.append(next_amount)
            possible_amounts = next_possible_amounts
        return -1
```

```python
class Solution:
    def coinChange(self, coins: List[int], amount: int) -> int:
        @cache
        def coin_change_helper(amount):
            if amount == 0:
                return 0
            if amount < 0:
                return -1
            def generate_possible():
                for coin in coins:
                    num_coins = coin_change_helper(amount - coin)
                    if num_coins == -1:
                        continue
                    yield num_coins + 1
            return min(generate_possible(), default=-1)
        return coin_change_helper(amount)
```

```python
class Solution:
    def coinChange(self, coins: List[int], amount: int) -> int:
        @cache
        def coin_change_helper(amount):
            if amount == 0:
                return 0
            if amount < 0:
                return -1
            
            def possible_coin():
                for coin in coins:
                    num_coins = coin_change_helper(amount - coin)
                    if num_coins == -1:
                        continue
                    yield num_coins + 1

            return min(possible_coin(), default=-1)
        
        return coin_change_helper(amount)
```
