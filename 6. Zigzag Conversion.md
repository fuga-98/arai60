```python
class Solution:
    def convert(self, s: str, numRows: int) -> str:
        zigzag = [['' for _ in range(len(s))] for _ in range(numRows)]
        s_i = 0
        s_length = len(s)
        column = 0
        while s_i < s_length:
            # down
            row = 0
            while row < numRows and s_i < s_length:
                zigzag[row][column] = s[s_i]
                row += 1
                s_i += 1
            # move diagonally
            row = numRows - 2
            column += 1
            while row > 0 and s_i < s_length:
                zigzag[row][column] = s[s_i]
                s_i += 1
                row -= 1
                column += 1
        
        making = ['' for _ in range(numRows)] 
        for i, elements in enumerate(zigzag):
            making[i] = ''.join(elements)
        return ''.join(making)

```

```python
class Solution:
    def convert(self, s: str, numRows: int) -> str:
        s_length = len(s)
				zigzag = [['' for _ in range(len(s))] for _ in range(numRows)]
        def move_down(s_i, column):
            row = 0
            while row < numRows and s_i < s_length:
                zigzag[row][column] = s[s_i]
                row += 1
                s_i += 1
            return s_i
        
        def move_diagonally(s_i, column):
            row = numRows - 2
            column += 1
            while row > 0 and s_i < s_length:
                zigzag[row][column] = s[s_i]
                s_i += 1
                row -= 1
                column += 1
            return s_i, column

        
        s_i = 0
        column = 0
        while s_i < s_length:
            s_i = move_down(s_i, column)
            s_i, column = move_diagonally(s_i, column)
            
        return ''.join(chain.from_iterable(zigzag))
```

けっこう満足。

### 3回連続で再現

```python
class Solution:
    def convert(self, s: str, numRows: int) -> str:
        zigzag = [[] for _ in range(numRows)]

        def move_down(index):
            row = 0
            while index < len(s) and row < numRows:
                zigzag[row].append(s[index])
                row += 1
                index += 1
            return index
        
        def move_up(index):
            row = numRows - 2
            while index < len(s) and row > 0:
                zigzag[row].append(s[index])
                row -= 1
                index += 1
            return index
        
        index = 0
        while index < len(s):
            index = move_down(index)
            index = move_up(index)
        return ''.join(c for row in zigzag for c in row)
```
